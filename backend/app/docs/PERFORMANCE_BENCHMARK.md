### 3. 缓存命中率测试

```python
import requests
import time

# 发送相同查询10次
query = {"query": "What is Bitcoin?"}
times = []

for i in range(10):
    start = time.time()
    r = requests.post("http://localhost:8000/api/v1/chat/quick-chat", json=query)
    elapsed = time.time() - start
    times.append(elapsed)
    print(f"Request {i+1}: {elapsed:.2f}s")

# 计算缓存命中效果
uncached = times[0]
cached_avg = sum(times[1:]) / len(times[1:])
improvement = (uncached - cached_avg) / uncached * 100
print(f"\nCache improvement: {improvement:.1f}%")
```
